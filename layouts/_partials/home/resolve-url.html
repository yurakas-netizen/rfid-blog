{{- $url := strings.TrimSpace (default "/" .Url) -}} {{- $lang := .Lang -}} {{- $langPrefix := printf "/%s/" $lang -}} {{- $result := "" -}} {{- if or (strings.HasPrefix $url "http://") (strings.HasPrefix $url "https://") (strings.HasPrefix $url "mailto:") (strings.HasPrefix $url "tel:") -}} {{- $result = $url -}} {{- else if and (ne $lang "uk") (or (eq $url (printf "/%s" $lang)) (strings.HasPrefix $url $langPrefix)) -}} {{- $result = $url -}} {{- else -}} {{- $trimmed := strings.TrimPrefix "/" $url -}} {{- $result = relLangURL $trimmed -}} {{- end -}} {{- return $result -}}
